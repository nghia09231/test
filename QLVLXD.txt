-- Bảng khách hàng
CREATE TABLE KhachHang (
    MaKH INT PRIMARY KEY IDENTITY,
    TenKH NVARCHAR(100),
    DiaChi NVARCHAR(200),
    SoDienThoai NVARCHAR(15),
    Email NVARCHAR(100)
);

-- Bảng vật liệu
CREATE TABLE VatLieu (
    MaVL INT PRIMARY KEY IDENTITY,
    TenVL NVARCHAR(100),
    DonViTinh NVARCHAR(50),
    GiaNhap DECIMAL(18, 2),
    GiaBan DECIMAL(18, 2),
    TonKho INT
);

-- Bảng nhập hàng
CREATE TABLE PhieuNhap (
    MaPN INT PRIMARY KEY IDENTITY,
    NgayNhap DATE,
    NhaCungCap NVARCHAR(100),
    TongTien DECIMAL(18, 2)
);

CREATE TABLE ChiTietPhieuNhap (
    MaPN INT,
    MaVL INT,
    SoLuong INT,
    DonGia DECIMAL(18,2),
    PRIMARY KEY (MaPN, MaVL),
    FOREIGN KEY (MaPN) REFERENCES PhieuNhap(MaPN),
    FOREIGN KEY (MaVL) REFERENCES VatLieu(MaVL)
);

-- Bảng hóa đơn bán hàng
CREATE TABLE HoaDon (
    MaHD INT PRIMARY KEY IDENTITY,
    MaKH INT,
    NgayBan DATE,
    TongTien DECIMAL(18,2),
    FOREIGN KEY (MaKH) REFERENCES KhachHang(MaKH)
);

CREATE TABLE ChiTietHoaDon (
    MaHD INT,
    MaVL INT,
    SoLuong INT,
    DonGia DECIMAL(18,2),
    PRIMARY KEY (MaHD, MaVL),
    FOREIGN KEY (MaHD) REFERENCES HoaDon(MaHD),
    FOREIGN KEY (MaVL) REFERENCES VatLieu(MaVL)
);

-- Bảng công nợ
CREATE TABLE CongNo (
    MaCN INT PRIMARY KEY IDENTITY,
    MaKH INT,
    NgayGhi DATE,
    SoTien DECIMAL(18,2),
    GhiChu NVARCHAR(200),
    FOREIGN KEY (MaKH) REFERENCES KhachHang(MaKH)
);
